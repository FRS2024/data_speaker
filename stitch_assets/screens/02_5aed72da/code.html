WITH cohort_base AS (
  SELECT 
    DATE_TRUNC(created_at, MONTH) AS cohort_month,
    account_id,

    SUM(mrr_amount) AS total_revenue,
    status
  FROM `corp-lakehouse-analytics.production_lake.fact_subscriptions`
  WHERE created_at >= '2024-01-01'
  GROUP BY 1, 2, 3, 5
)
SELECT cohort_month, tier, COUNT(DISTINCT account_id) AS active_accounts,
       SUM(total_revenue) AS net_mrr
FROM cohort_base
GROUP BY 1, 2 ORDER BY 1 DESC;
