WITH cohort_items AS (
SELECT
DATE_TRUNC(created_at, MONTH) AS cohort_month,
user_id,
mrr_amount,
plan_tier
FROM `production_lake.analytics_core.fact_subscriptions`
WHERE status = 'active' AND created_at >= '2024-01-01'
)
SELECT
cohort_month,
COUNT(DISTINCT user_id) AS active_subscribers,
SUM(mrr_amount) AS total_mrr,
AVG(mrr_amount) AS arpu,
ROUND(SUM(mrr_amount) / NULLIF(LAG(SUM(mrr_amount)) OVER (ORDER BY cohort_month), 0) - 1, 4) * 100 AS mom_growth_pct
FROM cohort_items
GROUP BY cohort_month
ORDER BY cohort_month DESC;
